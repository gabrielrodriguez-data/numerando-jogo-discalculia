package br.com.senac.pressao; import java.awt.BorderLayout; import java.awt.Component; import java.awt.Container; import java.awt.Dimension; import java.awt.FlowLayout; import java.awt.Font; import java.awt.GridBagConstraints; import javax.swing.table.DefaultTableModel; public class TelaMedicoes extends JFrame { private static final Font FONTE_GRANDE = new Font("SansSerif", Font.PLAIN, 18); private static final Font FONTE_TITULO = new Font("SansSerif", Font.BOLD, 24); private static final Font FONTE_CABECALHO = new Font("SansSerif", Font.BOLD, 16); private final ArquivoMedicoes arquivo = new ArquivoMedicoes(); private final JTextField txtData = new JTextField(14); private final JTextField txtHora = new JTextField(10); private final JTextField txtSistolica = new JTextField(10); private final JTextField txtDiastolica = new JTextField(10); private final JCheckBox chkEstresse = new JCheckBox("Estou em situação de estresse"); private final JButton btnSalvar = new JButton("Salvar"); private final JButton btnLimpar = new JButton("Limpar"); private final JButton btnFechar = new JButton("Fechar"); private final DefaultTableModel modelo = new DefaultTableModel( new Object[] {"Data", "Hora", "Pressão Sistólica", "Pressão Diastólica", "Estresse"}, 0) { @Override public boolean isCellEditable(int row, int column) { return false; } }; private final JTable tabela = new JTable(modelo); // Rótulos atualizados com a unidade de medida (mmHg) private final JLabel[] labelsDaTela = new JLabel[] { new JLabel("Data:"), new JLabel("Hora:"), new JLabel("Pressão sistólica (mmHg):"), new JLabel("Pressão diastólica (mmHg):") }; public TelaMedicoes() { setTitle("Registro de Medições de Pressão"); setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); setSize(900, 650); setMinimumSize(new Dimension(780, 580)); setLocationRelativeTo(null); construirInterface(); configurarAcessibilidade(); configurarAtalhos(); carregarArquivo(); preencherDataHoraAtual(); } private void construirInterface() { JPanel principal = new JPanel(new BorderLayout(15, 15)); principal.setBorder(BorderFactory.createEmptyBorder(20, 20, 20, 20)); JLabel titulo = new JLabel("Registro de Medições de Pressão"); titulo.setFont(FONTE_TITULO); titulo.setToolTipText("Tela para registrar e consultar medições de pressão arterial."); principal.add(titulo, BorderLayout.NORTH); JPanel formulario = new JPanel(new GridBagLayout()); formulario.setBorder(BorderFactory.createTitledBorder( BorderFactory.createEtchedBorder(), "Nova medição" )); JLabel lblData = labelsDaTela[0]; JLabel lblHora = labelsDaTela[1]; JLabel lblSistolica = labelsDaTela[2]; JLabel lblDiastolica = labelsDaTela[3]; lblData.setLabelFor(txtData); lblHora.setLabelFor(txtHora); lblSistolica.setLabelFor(txtSistolica); lblDiastolica.setLabelFor(txtDiastolica); GridBagConstraints gbc = new GridBagConstraints(); gbc.insets = new Insets(8, 10, 8, 10); gbc.anchor = GridBagConstraints.WEST; gbc.fill = GridBagConstraints.HORIZONTAL; adicionarCampo(formulario, gbc, 0, lblData, txtData); adicionarCampo(formulario, gbc, 1, lblHora, txtHora); adicionarCampo(formulario, gbc, 2, lblSistolica, txtSistolica); adicionarCampo(formulario, gbc, 3, lblDiastolica, txtDiastolica); gbc.gridx = 1; gbc.gridy = 4; gbc.gridwidth = 2; formulario.add(chkEstresse, gbc); JPanel botoes = new JPanel(new FlowLayout(FlowLayout.LEFT, 10, 5)); botoes.add(btnSalvar); botoes.add(btnLimpar); botoes.add(btnFechar); gbc.gridy = 5; formulario.add(botoes, gbc); JPanel topo = new JPanel(new BorderLayout()); topo.add(formulario, BorderLayout.CENTER); principal.add(topo, BorderLayout.CENTER); tabela.setFont(FONTE_GRANDE); tabela.setRowHeight(34); tabela.getTableHeader().setFont(FONTE_CABECALHO); tabela.setSelectionMode(ListSelectionModel.SINGLE_SELECTION); tabela.setFillsViewportHeight(true); JScrollPane scroll = new JScrollPane(tabela); scroll.setBorder(BorderFactory.createTitledBorder("Histórico de medições")); principal.add(scroll, BorderLayout.SOUTH); scroll.setPreferredSize(new Dimension(850, 280)); btnSalvar.addActionListener(e -> salvar()); btnLimpar.addActionListener(e -> limpar()); btnFechar.addActionListener(e -> dispose()); getRootPane().setDefaultButton(btnSalvar); add(principal); } private void adicionarCampo(JPanel painel, GridBagConstraints gbc, int linha, JLabel label, JTextField campo) { gbc.gridx = 0; gbc.gridy = linha; gbc.gridwidth = 1; gbc.weightx = 0; painel.add(label, gbc); gbc.gridx = 1; gbc.weightx = 1; campo.setPreferredSize(new Dimension(260, 38)); painel.add(campo, gbc); } private void configurarAcessibilidade() { for (JLabel label : labelsDaTela) { label.setFont(FONTE_GRANDE); } txtData.setFont(FONTE_GRANDE); txtHora.setFont(FONTE_GRANDE); txtSistolica.setFont(FONTE_GRANDE); txtDiastolica.setFont(FONTE_GRANDE); chkEstresse.setFont(FONTE_GRANDE); // Destaque visual no botão Salvar (Ação Principal) btnSalvar.setFont(new Font("SansSerif", Font.BOLD, 20)); btnSalvar.setBackground(new java.awt.Color(220, 245, 220)); btnSalvar.setForeground(new java.awt.Color(0, 100, 0)); btnLimpar.setFont(FONTE_GRANDE); btnFechar.setFont(FONTE_GRANDE); txtData.setToolTipText("Digite a data da medição no formato dd/MM/aaaa."); txtHora.setToolTipText("Digite o horário no formato HH:mm."); txtSistolica.setToolTipText("Digite somente números inteiros para a pressão sistólica em mmHg."); txtDiastolica.setToolTipText("Digite somente números inteiros para a pressão diastólica em mmHg."); chkEstresse.setToolTipText("Marque se você estava em uma situação de estresse no momento da medição."); btnSalvar.setToolTipText("Salva a medição no arquivo CSV. Atalho: Alt+S."); btnLimpar.setToolTipText("Limpa os campos da nova medição. Atalho: Alt+L."); btnFechar.setToolTipText("Fecha o sistema. Atalho: Alt+F."); btnSalvar.setMnemonic(KeyEvent.VK_S); btnLimpar.setMnemonic(KeyEvent.VK_L); btnFechar.setMnemonic(KeyEvent.VK_F); chkEstresse.setMnemonic(KeyEvent.VK_E); } private void configurarAtalhos() { JRootPane raiz = getRootPane(); raiz.getInputMap(JComponent.WHEN_IN_FOCUSED_WINDOW) .put(KeyStroke.getKeyStroke(KeyEvent.VK_ESCAPE, 0), "fechar"); raiz.getActionMap().put("fechar", new AbstractAction() { @Override public void actionPerformed(ActionEvent e) { dispose(); } }); raiz.getInputMap(JComponent.WHEN_IN_FOCUSED_WINDOW) .put(KeyStroke.getKeyStroke(KeyEvent.VK_S, KeyEvent.CTRL_DOWN_MASK), "salvarCtrl"); raiz.getActionMap().put("salvarCtrl", new AbstractAction() { @Override public void actionPerformed(ActionEvent e) { salvar(); } }); setFocusTraversalPolicy(new OrdemFocoPolicy( txtData, txtHora, txtSistolica, txtDiastolica, chkEstresse, btnSalvar, btnLimpar, btnFechar )); setFocusTraversalPolicyProvider(true); } private void preencherDataHoraAtual() { java.time.LocalDate hoje = java.time.LocalDate.now(); java.time.LocalTime agora = java.time.LocalTime.now(); txtData.setText(hoje.format(java.time.format.DateTimeFormatter.ofPattern("dd/MM/yyyy"))); txtHora.setText(agora.format(java.time.format.DateTimeFormatter.ofPattern("HH:mm"))); } private void carregarArquivo() { try { List medicoes = arquivo.carregar(); modelo.setRowCount(0); for (Medicao medicao : medicoes) { adicionarNaTabela(medicao); } } catch (IOException ex) { JOptionPane.showMessageDialog( this, "Não foi possível ler o arquivo de medições.\n" + "O programa continuará funcionando e poderá criar um novo arquivo.\n\n" + "Detalhes: " + ex.getMessage(), "Aviso", JOptionPane.WARNING_MESSAGE ); } catch (Exception ex) { JOptionPane.showMessageDialog( this, "Ocorreu um problema ao carregar o histórico.\n" + "Você ainda pode realizar novas medições.", "Aviso", JOptionPane.WARNING_MESSAGE ); } } private void salvar() { try { String data = Validador.validarData(txtData.getText()); String hora = Validador.validarHora(txtHora.getText()); int sistolica = Validador.validarPressao( txtSistolica.getText(), "sistólica" ); int diastolica = Validador.validarPressao( txtDiastolica.getText(), "diastólica" ); Medicao medicao = new Medicao( data, hora, sistolica, diastolica, chkEstresse.isSelected() ); arquivo.adicionar(medicao); adicionarNaTabela(medicao); JOptionPane.showMessageDialog( this, "Medição salva com sucesso!\n" + "O registro foi gravado no arquivo medicoes.csv.", "Sucesso", JOptionPane.INFORMATION_MESSAGE ); limparCamposMantendoDataHora(); } catch (ValidacaoException ex) { JOptionPane.showMessageDialog( this, ex.getMessage(), "Dados inválidos", JOptionPane.WARNING_MESSAGE ); } catch (IOException ex) { JOptionPane.showMessageDialog( this, "Não foi possível salvar a medição no arquivo.\n\n" + "Verifique se a pasta permite gravação.\n" + "Detalhes: " + ex.getMessage(), "Erro ao salvar", JOptionPane.ERROR_MESSAGE ); } catch (Exception ex) { JOptionPane.showMessageDialog( this, "Ocorreu um erro inesperado ao salvar a medição.\n" + "Nenhuma ação adicional foi realizada.", "Erro", JOptionPane.ERROR_MESSAGE ); } } private void adicionarNaTabela(Medicao medicao) { modelo.addRow(new Object[] { medicao.getData(), medicao.getHora(), medicao.getSistolica(), medicao.getDiastolica(), medicao.isEstresse() ? "Sim" : "Não" }); if (modelo.getRowCount() > 0) { tabela.scrollRectToVisible( tabela.getCellRect(modelo.getRowCount() - 1, 0, true) ); } } private void limpar() { txtData.setText(""); txtHora.setText(""); txtSistolica.setText(""); txtDiastolica.setText(""); chkEstresse.setSelected(false); txtData.requestFocusInWindow(); } private void limparCamposMantendoDataHora() { txtSistolica.setText(""); txtDiastolica.setText(""); chkEstresse.setSelected(false); txtSistolica.requestFocusInWindow(); } private static class OrdemFocoPolicy extends java.awt.FocusTraversalPolicy { private final List componentes; OrdemFocoPolicy(Component... componentes) { this.componentes = List.of(componentes); } @Override public Component getComponentAfter(Container aContainer, Component aComponent) { int indice = componentes.indexOf(aComponent); if (indice < 0) { return componentes.get(0); } return componentes.get((indice + 1) % componentes.size()); } @Override public Component getComponentBefore(Container aContainer, Component aComponent) { int indice = componentes.indexOf(aComponent); if (indice < 0) { return componentes.get(0); } return componentes.get((indice - 1 + componentes.size()) % componentes.size()); } @Override public Component getFirstComponent(Container aContainer) { return componentes.get(0); } @Override public Component getLastComponent(Container aContainer) { return componentes.get(componentes.size() - 1); } @Override public Component getDefaultComponent(Container aContainer) { return componentes.get(0); } } }
